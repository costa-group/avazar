# Fresh SMT variables after a call to a `func` (evaluated concretely).
# Expected: %x and %c get different SMT variables, and the formula holds
# iff %x = 1.

func @one() -> %r: ff {
  %r = 1
}

def @check(%x: ff) -> {
  call @one() to %p
  %c = bool.eq %p %x
  assert %c
}

def main(%x: ff) -> {
  call @check(%x) to
}
